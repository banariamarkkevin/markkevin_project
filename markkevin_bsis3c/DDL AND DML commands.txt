CREATE DATABASE IF NOT EXISTS kevinbsis3c;

USE kevinbsis3c;

CREATE TABLE IF NOT EXISTS employee (
		  EmplyeeID INT NOT NULL PRIMARY KEY,
		  FirstName VARCHAR(50) NOT NULL,
		  LastName VARCHAR(50) NOT NULL,
		  Position VARCHAR(50)NOT NULL
);
CREATE TABLE IF NOT EXISTS department (
		  DepartmentID INT NOT NULL PRIMARY KEY,
		  DepartmentName VARCHAR(50) NOT NULL
);
CREATE TABLE IF NOT EXISTS project (
		  ProjectID INT NOT NULL PRIMARY KEY,
		  ProjectName VARCHAR(50) NOT NULL,
		  DepartmentId VARCHAR(50) NOT NULL  
);
CREATE TABLE IF NOT EXISTS attendance (
		  AttendanceID INT NOT NULL PRIMARY KEY,
		  EmployeeID VARCHAR(50) NOT NULL,
		  Date VARCHAR(50) NOT NULL,
		 Status VARCHAR(50) NOT NULL	
);
CREATE TABLE IF NOT EXISTS salary (
		  SalaryID INT NOT NULL PRIMARY KEY,
		  EmployeeID VARCHAR(50) NOT NULL,
		  MonthlySalary VARCHAR(50) NOT NULL,
		  PayDate VARCHAR(50) NOT NULL	
);
